cmake_minimum_required(VERSION 3.16)
project(freertos_observatory VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(ENABLE_WARNINGS_AS_ERRORS "Treat compiler warnings as errors" OFF)
add_library(observatory_core src/observatory.c)
target_include_directories(observatory_core PUBLIC include)
if(MSVC)
 target_compile_options(observatory_core PRIVATE /W4)
else()
 target_compile_options(observatory_core PRIVATE -Wall -Wextra -Wpedantic)
 if(ENABLE_WARNINGS_AS_ERRORS)
  target_compile_options(observatory_core PRIVATE -Werror)
 endif()
endif()
add_executable(observatory_demo simulator/main.c)
target_link_libraries(observatory_demo PRIVATE observatory_core)
enable_testing()
add_executable(test_observatory tests/test_observatory.c)
target_link_libraries(test_observatory PRIVATE observatory_core)
add_test(NAME observatory_unit_tests COMMAND test_observatory)
